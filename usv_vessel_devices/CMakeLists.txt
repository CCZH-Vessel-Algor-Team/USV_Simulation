cmake_minimum_required(VERSION 3.16)
project(usv_vessel_devices)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-common5 REQUIRED)
find_package(gz-math7 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(vrx_gz REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/DeviceState.msg"
  "msg/ThrusterTiltState.msg"
  "srv/SetMastPosition.srv"
  "srv/SetThrusterTilt.srv"
  DEPENDENCIES std_msgs
)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)

add_library(VesselDevices SHARED src/VesselDevices.cc)
target_include_directories(VesselDevices PRIVATE include)
ament_target_dependencies(VesselDevices rclcpp std_msgs std_srvs)
target_link_libraries(VesselDevices gz-sim8::gz-sim8 gz-plugin2::register ${typesupport})
add_library(ThrusterTilt SHARED src/ThrusterTilt.cc)
target_include_directories(ThrusterTilt PRIVATE include)
ament_target_dependencies(ThrusterTilt rclcpp std_msgs)
target_link_libraries(ThrusterTilt gz-sim8::gz-sim8 gz-plugin2::register
  gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 vrx_gz::Waves ${typesupport})
target_include_directories(ThrusterTilt PRIVATE ${vrx_gz_DIR}/../../../include)
install(TARGETS VesselDevices ThrusterTilt LIBRARY DESTINATION lib)
install(DIRECTORY description config launch worlds docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md CHANGELOG.md DESTINATION share/${PROJECT_NAME})
install(FILES tools/convert_cad_assets.py DESTINATION share/${PROJECT_NAME}/tools)
ament_environment_hooks(env-hooks/vessel_devices.dsv.in)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_mast_control test/test_mast_control.cc)
  target_include_directories(test_mast_control PRIVATE include)
  ament_add_gtest(test_thruster_tilt_control test/test_thruster_tilt_control.cc)
  target_include_directories(test_thruster_tilt_control PRIVATE include)
  ament_add_gtest(test_propulsion_physics test/test_propulsion_physics.cc)
  target_include_directories(test_propulsion_physics PRIVATE include)
  ament_add_gtest(test_lamp_visual test/test_lamp_visual.cc)
  target_include_directories(test_lamp_visual PRIVATE include)
  ament_add_gtest(test_lamp_names test/test_lamp_names.cc)
  target_include_directories(test_lamp_names PRIVATE include)
  ament_add_pytest_test(test_model test/test_model.py)
  ament_add_pytest_test(test_cad_assets test/test_cad_assets.py)
  ament_add_pytest_test(test_simulation test/test_simulation.py TIMEOUT 180)
  ament_add_pytest_test(test_thruster_tilt_simulation test/test_thruster_tilt_simulation.py TIMEOUT 180)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
